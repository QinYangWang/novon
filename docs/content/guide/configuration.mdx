```ts title="novon.config.ts"
import { defineConfig } from 'novon/config'

export default defineConfig({
  title: 'Acme',                  // required
  description: 'Docs for Acme.',
  url: 'https://acme.dev',        // absolute links in sitemap and feeds
  base: '/',                      // serve from a sub-path, e.g. '/docs/'
  lang: 'en',
  layout: 'docs',                 // 'docs' | 'blog'
  layers: [],                     // see Layouts
  nav: [{ label: 'Guide', href: '/guide' }],
  theme: {
    accent: 'blue',               // blue, neutral, green, orange, rose, violet
    radius: '0.75rem',
    toc: true,
    logo: '/logo.svg',
    logoDark: '/logo-dark.svg',
    favicon: '/favicon.svg',
    social: { github: 'https://github.com/acme', email: 'hi@acme.dev' },
    editLink: 'https://github.com/acme/site/edit/main/',
    footer: { text: '© Acme', links: [{ label: 'Privacy', href: '/privacy' }] },
    authors: { 'Ada Park': { bio: 'Writes the changelog.', avatar: '/ada.jpg' } },
    css: ['./styles/extra.css'],
    override: { Footer: './theme/Footer.tsx' },
  },
  components: { Pricing: './components/Pricing.tsx' },
  plugins: ['search', 'sitemap', 'rss', 'llms'],
  og: { font: './fonts/NotoSansSC-Bold.ttf' }, // see Share images; false turns them off
  mdx: {
    highlight: { light: 'github-light', dark: 'github-dark' },
    remarkPlugins: [],
    rehypePlugins: [],
  },
})
```

## Share images

Every page gets a 1200×630 card at `/og/<route>.png` with its title, the site
name and the author's avatar or the site logo, plus the `og:image` and
`twitter:*` tags that point at it. Set `url` so the image link is absolute.

```ts
og: {
  enabled: true,               // or og: false
  font: './fonts/Sans.ttf',    // local TTF, OTF or WOFF; or ['./Regular.ttf', './Bold.ttf']
  family: 'Noto Sans SC',      // web font when font is unset; follows lang by default
  mode: 'light',               // 'light' | 'dark'
  accent: 'blue',              // defaults to theme.accent
}
```

Without `font`, each card downloads a subset of the web font holding only its
own characters, so CJK titles render without a large font file. Builds without
network need `font`; if neither works, pages ship without a card. Cards and
font subsets are cached in `.novon/og/`, so unchanged pages are not redrawn.
A page can use its own image or opt out in [frontmatter](/guide/content#frontmatter).

## Plugins

Built-in plugins are named by string. Your own plugin is an object with a
`buildEnd` hook that runs after every page is written:

```ts
const robots = {
  name: 'humans',
  async buildEnd({ emit, pages }) {
    await emit('humans.txt', `${pages.length} pages, written by people.`)
  },
}
```
